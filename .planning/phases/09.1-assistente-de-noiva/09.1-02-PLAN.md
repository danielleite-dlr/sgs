    isAssistantEligible(g):
    - noiva/studio/active/pacote marcado não excluído -> true
    - locationKind 'external' -> false (D-03: externa não tem assistente)
    - eventKind 'debutante' -> false (fora da fase)
    - status 'cancelled' -> false
    - pkg null -> false; pkg.includesBridalAssistant false -> false; pkg.deletedAt preenchido -> false

    assistantIneligibilityReason(g): null quando elegível; senão, na ordem de checagem:
    - status != active -> 'o grupo não está ativo'
    - eventKind != noiva -> 'só noivas têm assistente'
    - locationKind != studio -> 'evento fora do estúdio'
    - sem pacote / pacote excluído / sem flag -> 'o pacote não inclui assistente da noiva'

    isAssistantPending({ ...g, assistantMemberId, eventDateYmd, todayYmd, eventComandaClosed }):
    - elegível, assistantMemberId null, eventDateYmd '2026-12-05', todayYmd '2026-11-24', comanda não fechada -> true
    - mesmo com eventDateYmd == todayYmd -> true
    - eventDateYmd '2026-11-23' (passado) -> false
    - assistantMemberId preenchido -> false
    - eventComandaClosed true -> false
    - não elegível (externa) -> false

    assistantBlockNote('Maria Souza') -> 'Acompanhando noiva Maria Souza'
    assistantAccrualItemName('Noiva Maria') -> 'Assistente da Noiva — Noiva Maria'
    BRIDAL_ASSISTANT_BLOCK_KIND === 'acompanhamento'

    computePackageCosts:
    - { price: '1000.00', serviceCostCents: [30000], costItems: [{cost:'50.00'}], assistantDailyCents: 22000 } -> totalCost '570.00', profit '430.00', profitPercent '43.00'
