```ts
export type AccessSource = 'DEFAULT' | 'GRANTED' | 'DENIED';
export interface ResolvedItem { id: string; enabled: boolean; defaultEnabled: boolean; source: AccessSource; permissions: ReadonlySet<string>; }
export interface ResolvedAccess {
  roleName: string;
  rolePermissions: ReadonlySet<string>;
  items: ReadonlyMap<string, ResolvedItem>;
  /** rolePermissions ∪ managerPermissions dos itens GANHOS (para gates de UI e máscaras). */
  permissions: ReadonlySet<string>;
}
export const LOCKED_ROLES = ['ATTENDANT', 'PROFESSIONAL'] as const; // não recebem item grantable:false
export function defaultEnabledFor(item: AccessItemDef, rolePermissions: ReadonlySet<string>): boolean;
export function resolveAccess(input: { roleName: string; rolePermissions: Iterable<string>; overrides: ReadonlyArray<{ itemId: string; enabled: boolean }> }): ResolvedAccess;
export function expandGates(idsOrGates: readonly string[]): string[];
export type AccessCheck = { ok: true } | { ok: false; reason: 'ITEM_DISABLED' | 'MISSING_PERMISSIONS'; missing: string[] };
export function checkAccess(access: ResolvedAccess, gates: readonly string[], required: readonly string[]): AccessCheck;
export function isGranted(access: ResolvedAccess, itemId: string): boolean; // source === 'GRANTED'
export function hasCostVisibility(access: ResolvedAccess): boolean; // permissions tem package.write OU service.write OU financial.read (PERM-07)
```
Regras de `resolveAccess`: ADMIN → todos `enabled:true, source:'DEFAULT', permissions = rolePermissions`, overrides ignorados. Demais: `def = defaultEnabledFor`; override desconhecido ignorado; override com `enabled === def` → como sem override; `grantable:false` + papel em LOCKED_ROLES + override true → ignorado. `enabled && source DEFAULT` → `permissions = rolePermissions`; `GRANTED` → `rolePermissions ∪ managerPermissions`; desligado → `new Set()`.
`checkAccess`: `gates` vazio → `missing = required \ rolePermissions`; senão `candidates = expandGates(gates)`; nenhum candidato `enabled` → `ITEM_DISABLED` com `missing = required`; algum `enabled` com `required ⊆ item.permissions` → ok; senão `MISSING_PERMISSIONS` com o `missing` do primeiro candidato ligado.

Specs: `access-resolve.spec.ts` usa `ROLE_PERMISSIONS` como `rolePermissions` e um `it.each` com a matriz completa 4 papéis × 23 itens (tabela do `<behavior>`), mais os casos listados. `access-catalog.spec.ts` faz as checagens de catálogo e a paridade lendo menu-config.ts com `readFileSync` e uma regex que captura `id: '...'` cujo objeto contém `to:` antes do próximo `id:` (documentar a regex no spec; se o formato do arquivo impedir, parsear por blocos de `{ ... }`).
