export type TimelineKind = 'make' | 'hair';

export interface TimelineRuleStep {
  name: string;                 // "chegada - coffee", "Chegada fotógrafos"
  offsetMinutes: number;        // minutos antes da saída (0..1440)
  isExtra: boolean;             // extras (fotógrafos) vão depois da linha em branco
  kind: TimelineKind | null;    // só em etapa (não extra)
  startsRegistry: boolean;      // início do registro (fotos / making of)
  displayOrder: number;         // desempate
  durationMinutes: number | null; // só make/hair: quanto a profissional fica com a noiva (seed 90)
  serviceId?: string | null;    // serviço que vira agendamento da noiva
}

export interface TimelineInput {
  packageName: string;          // "Glam"
  brideLabel: string;           // "Vitoria" (o título sempre sai em maiúsculas)
  eventDateDdMm: string;        // "08/11"
  departureMinutes: number;     // 17:00 -> 1020
  steps: TimelineRuleStep[];    // etapas + extras do pacote
  makeProfessionalName?: string | null;
  hairProfessionalName?: string | null;
}

export interface TimelineRow {
  minutes: number;              // minutos do dia
  time: string;                 // formatHour(minutes)
  name: string;                 // nome da etapa como cadastrado
  label: string;                // texto da linha sem a hora ("make - profissional Camila")
  kind: TimelineKind | null;
  professionalName: string | null;
  isExtra: boolean;
  isRegistry: boolean;
  durationMinutes: number | null;
  serviceId: string | null;
}

export type TimelineWarningCode =
  | 'ARRIVAL_BEFORE_06' | 'PROFESSIONAL_UNDEFINED' | 'NO_REGISTRY_STEP' | 'SERVICE_UNDEFINED';
export interface TimelineWarning { code: TimelineWarningCode; message: string }

export interface TimelineResult {
  rows: TimelineRow[];          // etapas (ordem cronológica) seguidas dos extras (ordem cronológica)
  text: string;                 // texto para WhatsApp
  registryStartMinutes: number | null;
  registryName: string | null;  // "fotos" | "making of"
  warnings: TimelineWarning[];
}

export type TimelineDomainErrorCode = 'TIMELINE_RULES_MISSING' | 'BEFORE_MIDNIGHT' | 'INVALID_DEPARTURE';
export class TimelineDomainError extends Error {
  constructor(public readonly code: TimelineDomainErrorCode, message: string) { super(message); }
}

export function formatHour(totalMinutes: number): string;
export function parseDepartureTime(hhmm: string): number;        // lança INVALID_DEPARTURE
export function formatDdMm(isoDate: string): string;             // "2026-11-08" -> "08/11"
export function defaultBrideLabel(fullName: string): string;     // primeiro nome, toLocaleUpperCase('pt-BR')
export function computeTimeline(input: TimelineInput): TimelineResult;
export function brideWindows(result: TimelineResult): {
  kind: TimelineKind; serviceId: string | null; startMinutes: number; endMinutes: number;
}[];                                                              // end = start + durationMinutes (D-06: 90 no seed)
